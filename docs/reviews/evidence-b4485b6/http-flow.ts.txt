const sha = (b: Uint8Array) => createHash('sha256').update(b).digest('hex');
async function call(url: string, init: RequestInit = {}, status = 200) {
  const r = await fetch(origin + url, { ...init, signal: AbortSignal.timeout(25000), headers: { ...(init.method && init.method !== 'GET' ? { origin } : {}), ...init.headers } });
  const bytes = Buffer.from(await r.arrayBuffer());
  let body: any; try { body = JSON.parse(bytes.toString()); } catch { body = undefined; }
  evidence.results.push({ method: init.method ?? 'GET', url, status: r.status, expected: status, noStore: r.headers.get('cache-control'), error: body?.error?.code });
  assert.equal(r.status, status, `${init.method ?? 'GET'} ${url}: ${bytes.toString().slice(0,300)}`);
  return { body, bytes, headers: r.headers };
}
async function state(): Promise<State> { return (await call('/api/workspace')).body; }
async function mutate(command: any, expectedRevision?: number, status = 200) {
  return (await call('/api/workspace', { method: 'POST', headers: {'content-type':'application/json'}, body: JSON.stringify({ expectedRevision: expectedRevision ?? (await state()).revision, command }) }, status)).body;
}
async function settled() {
  for (let i = 0; i < 100; i++) {
    const s = await state();
    if (!s.parseAttempts.some(p => ['pending','running'].includes(p.status))) return s;
    await new Promise(r => setTimeout(r, 100));
  }
  throw new Error('bounded parse polling timed out');
}
async function upload(filename: string, bytes: Uint8Array, documentId?: string, key?: string) {
  const meta = { projectId:'demo-project', ...(documentId ? {documentId} : {}), filename, claimedMime: filename.endsWith('.pdf') ? 'application/pdf' : filename.endsWith('.docx') ? 'application/vnd.openxmlformats-officedocument.wordprocessingml.document' : filename.endsWith('.md') ? 'text/markdown' : 'text/plain', title:'Infrastructure synthetic evidence', category:'Methods', owner:'Synthetic reviewer', expectedRevision:(await state()).revision, idempotencyKey:key ?? `infra-${Date.now()}` };
  const result = (await call('/api/documents', {method:'POST',headers:{'content-type':'application/octet-stream','x-upload-metadata':Buffer.from(JSON.stringify(meta)).toString('base64url')},body:new Uint8Array(bytes)},201)).body;
  await settled();
  return {...result, meta};
}
